\section*{authcite 動作確認テスト (authcite-test)}

\subsection*{1. 既定の挙動 (lang=ja, two=etal)}
\begin{itemize}
  \item 1人著者 [期待値: Hosseini[1]]: \authcite{hosseini2020}
  \item 2人著者 [期待値: Duら[2]]: \authcite{du2021}
  \item 3人著者 [期待値: Smithら[3]]: \authcite{smith2022}
  \item others  [期待値: Leeら[4]]: \authcite{lee2023}
  \item 著者なし [期待値: [5]]: \authcite{manual2024}
\end{itemize}

\subsection*{2. 2人共著連記 (two=both / both)}
\begin{itemize}
  \item two=both [期待値: DuとGu[2]]: \authcite[two=both]{du2021}
  \item 省略形 both [期待値: DuとGu[2]]: \authcite[both]{du2021}
\end{itemize}

\subsection*{3. 言語指定オプション (lang=... / 省略形 / etal=...)}
\begin{itemize}
  \item lang=ja [期待値: Duら[2]]: \authcite[lang=ja]{du2021}
  \item lang=en [期待値: Du et al. [2]]: \authcite[lang=en]{du2021}
  \item 省略形 ja [期待値: Duら[2]]: \authcite[ja]{du2021}
  \item 省略形 en [期待値: Du et al. [2]]: \authcite[en]{du2021}
  \item etal=ja [期待値: Duら[2]]: \authcite[etal=ja]{du2021}
  \item etal=en [期待値: Du et al. [2]]: \authcite[etal=en]{du2021}
  \item 組み合わせ en + both [期待値: Du and Gu [2]]: \authcite[en,both]{du2021}
  \item 任意単語 etal={ほか} [期待値: Duほか[2]]: \authcite[etal={ほか}]{du2021}
\end{itemize}
