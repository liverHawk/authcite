\subsection{Citation Examples}
\begin{itemize}
  \item Single author (Lamport): \authcite{latex}
  \item Two authors, default et al. (Itoh et al.): \authcite{article1}
  \item Two authors, both (Itoh and Goto): \authcite[two=both]{article1}
  \item Three authors (Goossens et al.): \authcite{companion}
  \item With others (Foley et al.): \authcite{book1}
  \item Inverted single author (Baraff): \authcite{inproceedings1}
\end{itemize}
